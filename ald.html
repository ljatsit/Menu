SQL Script ສຳລັບ Supabase (PostgreSQL)

ຄັດລອກ Code ດ້ານລຸ່ມນີ້ໄປວາງໃນ SQL Editor ຂອງ Supabase ແລ້ວກົດ Run ເພື່ອສ້າງຕາຕະລາງທັງ 3 ຕາຕະລາງ:

-- 1. ສ້າງຕາຕະລາງ user
CREATE TABLE IF NOT EXISTS public."user" (
    userid int4 PRIMARY KEY,
    name varchar NOT NULL,
    section varchar
);

-- 2. ສ້າງຕາຕະລາງ code
CREATE TABLE IF NOT EXISTS public.code (
    codeid int4 PRIMARY KEY,
    code varchar NOT NULL
);

-- 3. ສ້າງຕາຕະລາງ data (ເຊື່ອມໂຍງ userid ກັບຕາຕະລາງ user)
CREATE TABLE IF NOT EXISTS public.data (
    dataid int4 PRIMARY KEY,
    userid int4 REFERENCES public."user"(userid) ON DELETE CASCADE ON UPDATE CASCADE,
    code varchar NOT NULL,
    date varchar NOT NULL,
    comment text
);

-- 4. ເປີດໃຊ້ງານ RLS ແລະ ອະນຸຍາດການເຂົ້າເຖິງ (ສຳລັບ Public API)
ALTER TABLE public."user" ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.code ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.data ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow all access to user" ON public."user" FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all access to code" ON public.code FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all access to data" ON public.data FOR ALL USING (true) WITH CHECK (true);

-- 5. ເພີ່ມຂໍ້ມູນຕົວຢ່າງເບື້ອງຕົ້ນ (Optional)
